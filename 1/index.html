首次使用：在 Supabase 里建表（点开展开 SQL） 
-- 在 Supabase → SQL Editor 中执行一次
create table if not exists public.jy_uploads (
  id uuid primary key default gen_random_uuid(),
  client_record_id text,
  teacher_name text not null,
  campus text,
  summary text,
  note text,
  file_name text,
  status text,
  exercise_title text,
  score int,
  score_detail text,
  created_at timestamptz not null default now()
);
alter table public.jy_uploads enable row level security;
create policy "jy_uploads_insert" on public.jy_uploads for insert to anon with check (true);
create policy "jy_uploads_select" on public.jy_uploads for select to anon using (true);

-- 若表是早期建的、没有打分字段，在 SQL Editor 再执行一次下面三行：
-- alter table public.jy_uploads add column if not exists exercise_title text;
-- alter table public.jy_uploads add column if not exists score int;
-- alter table public.jy_uploads add column if not exists score_detail text;

说明：上述策略允许任何人读写该表，仅适合内部小范围链接。若链接会公开，请改为登录用户可读或自建后端。
